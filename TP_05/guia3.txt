                    <input type="submit" name="generarReporte" value="Generar Reporte">

                <?php
                if (isset($_POST['generarReporte']) && !empty($_POST['nombreCurso'])) {
                    try {
                        $curso = $_SESSION['cursos'][$_POST['nombreCurso']] ?? null;
                        if ($curso) {
                            $reporte = new ReporteCurso($curso);
                            $resumen = $reporte->generarResumen();
                            echo "<p>Reporte del curso {$_POST['nombreCurso']}:</p>";
                            echo "<p>Cantidad de alumnos: {$resumen['cantidad_alumnos']}</p>";
                            echo "<p>Promedio general: " . number_format($resumen['promedio_general'], 2) . "</p>";
                            echo "<p>Alumnos aprobados por materia:</p><ul>";
                            foreach ($resumen['aprobados_por_materia'] as $materia => $cantidad) {
                                echo "<li>$materia: $cantidad alumnos</li>";
                            }
                            echo "</ul>";
                        } else {
                            echo "<p>Error: El curso no existe.</p>";
                        }
                    } catch (Exception $e) {
                        echo "<p>Error: {$e->getMessage()}</p>";
                    }
                }

                if (isset($_POST['exportarReporte']) && !empty($_POST['nombreCurso'])) {
                    try {
                        $curso = $_SESSION['cursos'][$_POST['nombreCurso']] ?? null;
                        if ($curso) {
                            $reporte = new ReporteCurso($curso);
                            $reporte->exportarAArchivo('reporte_' . $_POST['nombreCurso'] . '.txt');
                            echo "<p>Reporte exportado a 'reporte_{$_POST['nombreCurso']}.txt'.</p>";
                        } else {
                            echo "<p>Error: El curso no existe.</p>";
                        }
                    } catch (Exception $e) {
                        echo "<p>Error: {$e->getMessage()}</p>";
                    }
                }
                ?>
